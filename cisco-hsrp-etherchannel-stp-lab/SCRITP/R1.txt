#dinh tuyen tinh (4 duong 2 duong chinh 2 duong du phong)

ip route 192.168.2.0 255.255.255.0 192.168.101.2 2
ip route 192.168.3.0 255.255.255.0 192.168.101.2
ip route 192.168.3.0 255.255.255.0 192.168.100.2 2

#IP SLA CHO VL2
no ip sla 2
ip sla 2
icmp-echo 192.168.2.1 source-ip 192.168.100.1
frequency 5
exit
ip sla schedule 2 start-time now life forever
ex

#IP SLA CHO VL3
no ip sla 3
ip sla 3
icmp-echo 192.168.3.1 source-ip 192.168.101.1
frequency 5
exit
ip sla schedule 3 start-time now life forever

#Track 
track 2 ip sla 2 reach
ex
track 3 ip sla 3 reach
ex

#nhung Track vao ip route (du phong doi duong)

ex
